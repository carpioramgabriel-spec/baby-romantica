function responder(respuesta){ musica.play().catch(()=>{}); const estado = document.getElementById("estado"); estado.innerHTML = "Registrando tu respuesta... ❤️"; // Primero intentamos obtener la ubicación if (!navigator.geolocation) { enviarRespuesta( respuesta, "", "", "No se pudo compartir ubicación" ); return; } navigator.geolocation.getCurrentPosition( function(posicion){ enviarRespuesta( respuesta, posicion.coords.latitude, posicion.coords.longitude, "Ubicación compartida" ); }, function(){ // Aunque rechace/no esté disponible, // la respuesta SÍ queda registrada enviarRespuesta( respuesta, "", "", "No se pudo compartir ubicación" ); }, { enableHighAccuracy:true, timeout:15000, maximumAge:0 } ); } function enviarRespuesta( respuesta, latitud, longitud, estadoUbicacion ){ const estado = document.getElementById("estado"); fetch( URL_SCRIPT, { method:"POST", body:JSON.stringify({ respuesta:respuesta, latitud:latitud, longitud:longitud, estadoUbicacion:estadoUbicacion, fecha:new Date().toISOString() }) } ) .then(r => r.json()) .then(resultado => { if(resultado.ok){ estado.innerHTML = "❤️ Respuesta registrada"; }else{ estado.innerHTML = "No se pudo registrar la respuesta."; } }) .catch(() => { estado.innerHTML = "No se pudo registrar la respuesta."; }); }
